World Flags

viewof selectedCountry = Inputs.select(
  flags_data.country,
  {
    label: "Select a country:",
    value: "United States"
  }
)
selectedFlag = {
  const idx = flags_data.country.indexOf(selectedCountry);
  return {
    name: selectedCountry,
    code: flags_data.code[idx],
    url: flags_data.url[idx]
  };
}

html`
  <div style="margin-top: 1.5rem;">
    <h3 style="margin-bottom: 0.75rem;">${selectedFlag.name}</h3>
    <img 
      src="${selectedFlag.url}" 
      alt="Flag of ${selectedFlag.name}" 
      style="max-width: 480px; width: 100%; height: auto; border: 1px solid #ddd; border-radius: 6px; box-shadow: 0 4px 10px rgba(0,0,0,0.12); display: block;"
    />
  </div>
`
